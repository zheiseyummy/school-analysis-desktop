    环境变量设置用户密码
    setx MINIO_ROOT_USER admin123
    setx MINIO_ROOT_PASSWORD <请设置本机密码>
    启动服务
    .\minio.exe server C:\storage\data --console-address "127.0.0.1:9000" --address "127.0.0.1:9005"



minio:
  config:
    url: http://127.0.0.1:9005 #ip地址
    accessKey: <本机账号> #  账号
    secretKey: <本机密码> #  密码
    secure: false #如果是true，则用的是https而不是http,默认值是true
    bucketName: "big-event"  # 桶的名字 相当于文件夹
